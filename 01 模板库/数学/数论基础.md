# 数论基础

> 集合：快速幂 / 龟速乘、exgcd、求逆元、线性筛（质数 / phi / mu）、分解质因数。

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

// 快速幂（模数 < 2^63 时用 mul 续接）
ll qpow(ll a, ll b, ll mod){
    ll r = 1;
    while(b){ if(b & 1) r = r * a % mod; a = a * a % mod; b >>= 1; }
    return r;
}
ll mul(ll a, ll b, ll mod){               // 龟速乘：防 1e18*1e18 爆 ll
    ll r = 0;
    while(b){ if(b & 1) r = (r + a) % mod; a = (a + a) % mod; b >>= 1; }
    return r;
}
ll exgcd(ll a, ll b, ll &x, ll &y){       // 返回 gcd，解 ax+by=gcd
    if(!b){ x = 1; y = 0; return a; }
    ll g = exgcd(b, a % b, y, x);
    y -= a / b * x;
    return g;
}
ll inv(ll x, ll mod){ ll a, b; exgcd(x, mod, a, b); return (a % mod + mod) % mod; } // 也可 a=mod-2 费马

// 线性筛：primes/phi/mu/minp 一把梭（O(n)）
const int N = 1e6 + 10;
vector<int> primes;
int phi[N], mu[N];
bool vis[N];
void sieve(int n){
    mu[1] = phi[1] = 1;
    for(int i = 2; i <= n; i++){
        if(!vis[i]){ primes.push_back(i); phi[i] = i - 1; mu[i] = -1; }
        for(int p : primes){
            if(1ll * i * p > n) break;
            vis[i * p] = true;
            if(i % p == 0){ phi[i * p] = phi[i] * p; mu[i * p] = 0; break; } // p 是最小质因子且整除
            else { phi[i * p] = phi[i] * (p - 1); mu[i * p] = -mu[i]; }
        }
    }
}

// 逆元线性递推：inv[i] = -(MOD/i)*inv[MOD%i] % MOD
vector<int> getInv(int n, int mod){
    vector<int> f(n + 1); f[1] = 1;
    for(int i = 2; i <= n; i++) f[i] = 1ll * (mod - mod / i) * f[mod % i] % mod;
    return f;
}

// 分解质因数（试除 + 尾项）
vector<pair<ll,int>> factor(ll n){
    vector<pair<ll,int>> f;
    for(ll p = 2; p * p <= n; p++)
        if(n % p == 0){ int c = 0; while(n % p == 0) n /= p, c++; f.push_back({p, c}); }
    if(n > 1) f.push_back({n, 1});
    return f;
}
```
