> 复杂度：build $O(n)$，单次 modify / query $O(\log n)$。开 4 倍空间。
> 适用：区间加 + 区间求和（懒标记）。改 long long 注意 `t` 结构体已用 `long long`，传入 `v` 需匹配。
> 改造提示：区间赋值/区间最值改 `add` 语义与 pushdown；根为 1，左右儿子 `p<<1` 与 `p<<1|1`。

```cpp
const int N = 1e5 + 10;
int a[N];
struct { int l, r; long long sum, add; } t[N << 2];
void build(int p, int l, int r) {
    t[p].l = l; t[p].r = r;
    if (l == r) { t[p].sum = a[l]; return; }
    int mid = (l + r) >> 1;
    build(p << 1, l, mid); build(p << 1 | 1, mid + 1, r);
    t[p].sum = t[p << 1].sum + t[p << 1 | 1].sum;
}
void pushdown(int p) {
    if (!t[p].add) return;
    for (int s : {p << 1, p << 1 | 1}) {
        t[s].sum += (t[s].r - t[s].l + 1) * t[p].add;
        t[s].add += t[p].add;
    }
    t[p].add = 0;
}
void modify(int p, int l, int r, long long v) {
    if (l <= t[p].l && t[p].r <= r) {
        t[p].sum += (t[p].r - t[p].l + 1) * v;
        t[p].add += v; return;
    }
    pushdown(p);
    int mid = (t[p].l + t[p].r) >> 1;
    if (l <= mid) modify(p << 1, l, r, v);
    if (r >  mid) modify(p << 1 | 1, l, r, v);
    t[p].sum = t[p << 1].sum + t[p << 1 | 1].sum;
}
long long query(int p, int l, int r) {
    if (l <= t[p].l && t[p].r <= r) return t[p].sum;
    pushdown(p);
    int mid = (t[p].l + t[p].r) >> 1;
    long long res = 0;
    if (l <= mid) res += query(p << 1, l, r);
    if (r >  mid) res += query(p << 1 | 1, l, r);
    return res;
}
```
