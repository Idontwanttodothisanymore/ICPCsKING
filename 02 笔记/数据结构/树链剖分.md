给树做一个 dfs 重编号，使**树上的任意一条路径都可以拆成 $\log n$ 段连续区间，任意一棵子树都是连续区间**，于是树上的问题变成的连续的区间问题，可以直接交给线段树，线段树神力！

##### 定义与概念

- 重儿子：子树最大的儿子，有多个任取一个；
- 轻儿子：其余的儿子；
- 重边：$x \to$ 重儿子的边；
- 重链：由重边首尾相连组成的链。链的顶端称为**链头**；

因此，轻儿子一定是某条重链的链头。
##### 主要性质

路径中 **轻边的数量 $\leq \log n$**，也就是说链的数量 $\leq \log n$。

从任意点 $u$ 走到 $n$，每经过一条轻边 $x\to p$，说明 $x$ 不是 $p$ 的重儿子，设 $x$ 的重儿子为 $y$，即有 $sz_{y}\geq sz_{x}$。所以 $sz_{p}\geq 1+sz_{y}+sz_{x}\geq 2\times sz_{x}$。即通过一条轻边，当前子树的大小至少翻倍。

因此，轻边的数量最多是 $\lfloor \log_{2}n \rfloor$。
##### 核心算法

树剖的预处理是两遍 dfs。

第一遍 dfs 求重儿子：算出父节点，子树大小，深度，重儿子；
第二步计算编号：优先递归重儿子，保证了他们拿到的是连续的 dfn。

- 路径操作：**总是把"链头深度更大"的那一侧整段重链处理掉，然后跳到链头的父亲 ，直到两点落到同一条重链上。**
  ![[无标题的笔记本-25 2.jpg]]

- 子树操作：天然连续，这是由 dfn 保证的。

##### 代码实现

树剖的主要难度还是在代码上。细节比较多，还有一些处理的 tricks。

其中的 rnk 为 dfn 的反函数，即对于任意的 $x$，有 `rnk[dfn[x]]=x`。


```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

const int N = 1e5 + 10;
int n, m, r, P;
int val[N];                 // 各节点初始权值（按节点编号存）
vector<int> e[N];

int fa[N], dep[N], sz[N], son[N];
int top[N], dfn[N], rnk[N], id;

ll sum[N << 2], tag[N << 2];

void dfs1(int x, int f) {                    // 第一遍：求 fa/dep/sz/重儿子
    fa[x] = f; dep[x] = dep[f] + 1; sz[x] = 1; son[x] = 0;
    for (int y : e[x]) {
        if (y == f) continue;
        dfs1(y, x);
        sz[x] += sz[y];
        if (sz[y] > sz[son[x]]) son[x] = y;  // sz[0]=0，叶子天然处理
    }
}

void dfs2(int x, int t) {                    // 第二遍：重儿子优先编号
    top[x] = t;
    dfn[x] = ++id;
    rnk[id] = x;
    if (son[x]) dfs2(son[x], t);             // 先走重儿子，保证重链 dfn 连续
    for (int y : e[x])
        if (y != fa[x] && y != son[x]) dfs2(y, y);   // 轻儿子开新链
}

void build(int x, int l, int r) {
    tag[x] = 0;
    if (l == r) { sum[x] = val[rnk[l]] % P; return; }  // 按 dfn 取权值
    int mid = (l + r) >> 1;
    build(x << 1, l, mid);
    build(x << 1 | 1, mid + 1, r);
    sum[x] = (sum[x << 1] + sum[x << 1 | 1]) % P;
}

void pd(int x, int l, int r) {
    if (!tag[x]) return;
    int mid = (l + r) >> 1;
    tag[x << 1] = (tag[x << 1] + tag[x]) % P;
    sum[x << 1] = (sum[x << 1] + tag[x] * (mid - l + 1)) % P;
    tag[x << 1 | 1] = (tag[x << 1 | 1] + tag[x]) % P;
    sum[x << 1 | 1] = (sum[x << 1 | 1] + tag[x] * (r - mid)) % P;
    tag[x] = 0;
}

void modify(int x, int l, int r, int ql, int qr, int k) {
    if (ql <= l && r <= qr) {                // 完全覆盖
        sum[x] = (sum[x] + (ll)k * (r - l + 1)) % P;
        tag[x] = (tag[x] + k) % P;
        return;
    }
    pd(x, l, r);
    int mid = (l + r) >> 1;
    if (ql <= mid) modify(x << 1, l, mid, ql, qr, k);
    if (qr > mid) modify(x << 1 | 1, mid + 1, r, ql, qr, k);
    sum[x] = (sum[x << 1] + sum[x << 1 | 1]) % P;
}

ll query(int x, int l, int r, int ql, int qr) {
    if (ql <= l && r <= qr) return sum[x];
    pd(x, l, r);
    int mid = (l + r) >> 1;
    ll res = 0;
    if (ql <= mid) res += query(x << 1, l, mid, ql, qr);
    if (qr > mid) res += query(x << 1 | 1, mid + 1, r, ql, qr);
    return res % P;
}

void upd_path(int u, int v, int k) {         // 路径加：沿链头向上跳
    while (top[u] != top[v]) {
        if (dep[top[u]] < dep[top[v]]) swap(u, v);
        modify(1, 1, n, dfn[top[u]], dfn[u], k);
        u = fa[top[u]];
    }
    if (dep[u] > dep[v]) swap(u, v);
    modify(1, 1, n, dfn[u], dfn[v], k);
}

ll qry_path(int u, int v) {                   // 路径和
    ll res = 0;
    while (top[u] != top[v]) {
        if (dep[top[u]] < dep[top[v]]) swap(u, v);
        res = (res + query(1, 1, n, dfn[top[u]], dfn[u])) % P;
        u = fa[top[u]];
    }
    if (dep[u] > dep[v]) swap(u, v);
    res = (res + query(1, 1, n, dfn[u], dfn[v])) % P;
    return res;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cin >> n >> m >> r >> P;
    for (int i = 1; i <= n; ++i) cin >> val[i];
    for (int i = 1; i < n; ++i) {
        int x, y; cin >> x >> y;
        e[x].push_back(y); e[y].push_back(x);
    }
    dfs1(r, 0);
    id = 0;
    dfs2(r, r);
    build(1, 1, n);
    while (m--) {
        int op, x, y, z;
        cin >> op;
        if (op == 1) { cin >> x >> y >> z; upd_path(x, y, z % P); }
        else if (op == 2) { cin >> x >> y; cout << qry_path(x, y) << '\n'; }
        else if (op == 3) { cin >> x >> z; modify(1, 1, n, dfn[x], dfn[x] + sz[x] - 1, z % P); }
        else { cin >> x; cout << query(1, 1, n, dfn[x], dfn[x] + sz[x] - 1) << '\n'; }
    }
    return 0;
}
```
